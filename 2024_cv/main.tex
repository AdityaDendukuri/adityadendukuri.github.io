\section*{Research Experience}
\entry{University of California, Santa Barbara}{Santa Barbara, CA}
{Graduate Research Assistant; advisor: Linda R. Petzold}{Sept. 2021 -- Present}
\begin{itemize}
\item Developed an adaptive method for stiff chemical master equations that uses probability flux to
select molecular states and time steps. Derived bounds for the errors introduced by truncation and
numerical propagation. Published in \textit{Multiscale Modeling \& Simulation} (2026).
\item Constructed multigrid transfer operators and coarse equations for the reaction--diffusion
master equation. Analyzed the error introduced by coarsening and computed transient distributions
with up to 7.16 million states.
\item Developed adjoint Krylov methods that reuse forward and adjoint approximations to compute
derivatives with respect to many unknown coefficients. Applied the methods to chemical master
equations, Kuramoto--Sivashinsky dynamics, and Lindblad equations.
\item Analyzed the bias introduced by projection in Koopman operator inference. Recovered ten
reaction rates from a $15\times15$ operator for a model with more than 146 million molecular states.
\item Contributed analyses of time series and reduced models for neuronal firing sequences in human
brain organoids (\textit{Nature Neuroscience}, 2026).
\end{itemize}

\entry{Lawrence Berkeley National Laboratory}{Berkeley, CA}
{UC Graduate Student Researcher Intern; mentor: Mark Fornace}{July -- September 2026}
\begin{itemize}
\item Developed methods that accelerate the simulation of continuous-time Markov chains by replacing
repeated transitions within a subnetwork with a calculation of where the process exits.
\item Implemented subnetwork sweeping in C++, using sparse factorizations and estimates of occupation
times to guide the calculation. A public release is planned.
\end{itemize}

\entry{NSF National Center for Atmospheric Research}{Boulder, CO}
{SIParCS Graduate Student Intern}{May -- July 2024}
\begin{itemize}
\item Developed C++ components of the TUV-x radiative-transfer code, including a tridiagonal solver.
Compared the solver with LAPACKE and added numerical tests and performance benchmarks.
\end{itemize}

\section*{Software}
\begin{itemize}
\item \href{https://github.com/AdityaDendukuri/numerics}{\textbf{numerics}}.
C++20 library for scientific computing, including sparse and structured factorizations, Krylov
methods, and matrix functions. Supports reuse of factorizations and LU factorization in mixed
precision, with numerical tests and performance benchmarks.
\item \href{https://github.com/AdityaDendukuri/markovkit}{\textbf{MarkovKit}}.
C++20 library implementing adaptive finite state projection, stochastic simulation,
and multigrid solvers for spatial reaction systems.
\item \href{https://github.com/AdityaDendukuri/AdjointKrylovGradients}{\textbf{AdjointKrylovGradients}}.
Julia package for computing gradients through adjoint Krylov approximations. An \texttt{rrule}
interface allows the method to be used in automatic differentiation. Includes code for reproducing
the numerical experiments.
\item \href{https://github.com/AdityaDendukuri/approxchol}{\textbf{approxchol}}.
Header-only C++ implementation of randomized approximate Cholesky factorization for graph Laplacians.
\end{itemize}
